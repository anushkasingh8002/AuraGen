import React, { useState } from "react";
import { Button, Input, Card, ProgressBar } from "@/ui";

export default function SimplifiedWizard({ initialValues = {}, onChange, onSubmit, onCancel }) {
  const [step, setStep] = useState(1);
  const [values, setValues] = useState({
    fullName: initialValues.fullName || "",
    email: initialValues.email || "",
  });

  const totalSteps = 3;

  const handleChange = (field, val) => {
    const updated = { ...values, [field]: val };
    setValues(updated);
    if (onChange) onChange(updated);
  };

  const handleNext = () => setStep((s) => Math.min(s + 1, totalSteps));
  const handleBack = () => setStep((s) => Math.max(s - 1, 1));

  const handleSubmit = (e) => {
    if (e && e.preventDefault) e.preventDefault();
    if (onSubmit) onSubmit(values);
  };

  return (
    <Card className="max-w-md mx-auto p-6 space-y-6">
      <div className="flex justify-between items-center text-sm text-gray-500">
        <span>Step {step} of {totalSteps}</span>
        <button type="button" onClick={onCancel} className="text-blue-600 hover:underline">
          Back to full form
        </button>
      </div>
      <ProgressBar current={step} total={totalSteps} />

      {step === 1 && (
        <div className="space-y-4">
          <h2 className="text-xl font-bold">What is your full legal name?</h2>
          <p className="text-sm text-gray-600">Please enter your name as it appears on official documents.</p>
          <Input
            id="fullName"
            label="Full Name"
            value={values.fullName}
            onChange={(e) => handleChange("fullName", e.target.value)}
          />
        </div>
      )}

      {step === 2 && (
        <div className="space-y-4">
          <h2 className="text-xl font-bold">Where should we contact you?</h2>
          <p className="text-sm text-gray-600">We will send your loan updates and status here.</p>
          <Input
            id="email"
            type="email"
            label="Email Address"
            value={values.email}
            onChange={(e) => handleChange("email", e.target.value)}
          />
        </div>
      )}

      {step === 3 && (
        <div className="space-y-4">
          <h2 className="text-xl font-bold">Review your details</h2>
          <div className="bg-gray-50 p-4 rounded text-sm space-y-2">
            <div><strong>Full Name:</strong> {values.fullName || "Not provided"}</div>
            <div><strong>Email:</strong> {values.email || "Not provided"}</div>
          </div>
        </div>
      )}

      <div className="flex justify-between pt-4 border-t">
        {step > 1 && (
          <Button variant="secondary" onClick={handleBack}>
            Back
          </Button>
        )}
        <div className="ml-auto">
          {step < totalSteps ? (
            <Button variant="primary" onClick={handleNext}>
              Next
            </Button>
          ) : (
            <Button variant="primary" onClick={handleSubmit}>
              Submit Application
            </Button>
          )}
        </div>
      </div>
    </Card>
  );
}
