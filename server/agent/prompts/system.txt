import React, { useState } from "react";
import { Button, Input, Card, ProgressBar } from "@/ui";
export default function Wizard({ initialValues = {}, onChange = () => {}, onSubmit = () => {}, onCancel = () => {} }) {
  const [step, setStep] = useState(1);
  const [values, setValues] = useState({ ...initialValues });
  const set = (k, v) => { const next = { ...values, [k]: v }; setValues(next); onChange(next); };
  return (
    <Card className="max-w-md mx-auto p-6 space-y-4">
      <div className="flex justify-between text-sm"><button type="button" onClick={onCancel}>Back to full form</button><span>Step {step}/2</span></div>
      <ProgressBar current={step} total={2} />
      {step === 1 && <Input id="field" label="Field" value={values.field || ""} onChange={(e) => set("field", e.target.value)} />}
      {step === 2 && <Button onClick={() => onSubmit(values)}>Submit</Button>}
    </Card>
  );
}
